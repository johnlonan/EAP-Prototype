CSS Usage

/* Add .custom-scrollbar to any scrollable element */ .custom-scrollbar { /* Firefox */ scrollbar-width: thin; scrollbar-color: rgba(0, 0, 0, 0.15) transparent; } .custom-scrollbar::-webkit-scrollbar { width: 12px; /* vertical track width */ height: 12px; /* horizontal track height */ } .custom-scrollbar::-webkit-scrollbar-track { background: transparent; } .custom-scrollbar::-webkit-scrollbar-thumb { background-color: rgba(0, 0, 0, 0.15); border-radius: 9999px; border: 4px solid transparent; /* creates 4px padding */ background-clip: padding-box; }
